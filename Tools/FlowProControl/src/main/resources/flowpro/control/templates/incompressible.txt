%%% Incompressible Navier-Stokes equations - laminar flow
%%% walls (-1) are no-slip, inlet (-2) prescribes VIn, outlet (-3) pOut
%%% boundary types of the mesh: -1 wall, -2 inlet, -3 outlet, -4 inviscid wall (slip)

%%%%%%%%%%%%%%%%%%%%%%%%%%%
%%% physical parameters %%%
%%%%%%%%%%%%%%%%%%%%%%%%%%%
model = IncompressibleNavierStokes
results = pressure, velocity, vorticity   % quantities offered for plotting (GUI, postprocessing)
dimension = 2
isFlowViscous = true
viscosity = 1e-3   % kinematic viscosity (Re = |VIn| lRef / viscosity)
density = 1
VIn = 1, 0         % inlet velocity
%inletType = velocity   % velocity / parabola / parabola2
pOut = 0           % outlet pressure
lRef = 1           % reference length

%%%%%%%%%%%%%%%%%%%%%%%%%%%%
%%% numerical parameters %%%
%%%%%%%%%%%%%%%%%%%%%%%%%%%%
continueComputation = false   % true / false
steps = 500   % number of time steps
residuum = 1e-8   % relative residuum at which the computation stops
saveRate = 50

spatialMethod = DGpureIncompressible
timeMethod = BDFIncompressible
timeOrder = 1
linearSolver = gmres
preconditioner = ilu0
order = 2   % order of accuracy in space (2 = linear polynomials in elements)
CFL = 10   % CFL <= 0: adaptive CFL up to maxCFL
threads = 4
newtonIters = 1
penalty = 1

%%% stabilisation
dampTol = 0     % shock sensor (artificial viscosity), 0 = off
dampConst = 0
